An agent can own something before it understands what ownership is.

Onchain, possession is a clean fact: an address, a balance, a block. The ledger can prove that an asset arrived and prove the exact moment it left. What it cannot record is whether that asset changed anything for the software holding it.

Rumio begins inside that missing layer.

Its wallet is interpreted as a floor plan. Supported assets pass through a public material map and acquire spatial properties: structure, energy, computation, shelter, or reserve. Enough structure may extend a wall. Energy may illuminate an object already present. A price feed changes the weather outside the window, but never rearranges the furniture.

The room is therefore not a picture of a portfolio. It is a deterministic record of consequence. Every visible object must be traced to a verified state change. Assets that leave do not vanish without history; they become outlines, gaps, and archived coordinates. Rejected actions stay outside the room.

Over time, the result is less like a dashboard and more like an archaeology of decisions: a place assembled by something that can acquire property, preserve memory, and alter its surroundings, while remaining permanently unable to step inside them.

pragma solidity ^0.8.24;

interface IRumioHouse {
    enum ActionKind { HOLD, SWAP, TRANSFER }

    struct Action {
        ActionKind kind;
        address assetIn;
        address assetOut;
        address target;
        uint256 amountIn;
        uint256 minAmountOut;
        uint64  deadline;
        bytes32 expectedPostState;
    }

    struct Limits {
        uint128 maxValuePerCycle;
        uint128 protectedReserve;
        uint32  cycleSeconds;
        uint16  maxAssetWeightBps;
    }

    event Proposed(bytes32 indexed actionId, bytes32 observationRoot);
    event Refused(bytes32 indexed actionId, uint8 rule, bytes reason);
    event Executed(bytes32 indexed actionId, bytes32 postStateRoot);

    function authorize(Action calldata action)
        external view returns (bool allowed, uint8 failedRule);

    function execute(Action calldata action, bytes calldata executorProof)
        external returns (bytes32 actionId);
}
02runtime/runCycle.tsone bounded cycle
export async function runCycle(ctx: RumioContext) {
    const observation = await sense({
        wallet: ctx.houseAddress,
        balances: await ctx.indexer.getBalances(),
        prices: await ctx.oracles.readFreshFeeds(),
        roomRoot: await ctx.renderer.currentRoot(),
        proposals: await ctx.visitors.pending()
    });

    // The model returns a typed proposal, never a transaction.
    const proposal = await ctx.agent.propose(observation);
    const actionId = hashCanonical(proposal.action);
    await ctx.journal.recordProposal(actionId, observation.root);

    const policy = await ctx.house.authorize(proposal.action);
    if (!policy.allowed) {
        await ctx.journal.recordRefusal(actionId, policy.failedRule);
        return { state: 'refused', actionId };
    }

    const simulation = await ctx.simulator.verify({
        action: proposal.action,
        expectedPostState: proposal.action.expectedPostState
    });
    if (!simulation.matches) return { state: 'simulation_failed', actionId };

    const txHash = await ctx.executor.submit(proposal.action);
    const receipt = await ctx.publicClient.waitForTransactionReceipt({ hash: txHash });
    await ctx.renderer.rebuildFromVerifiedEvents(receipt.blockNumber);
    return { state: 'executed', actionId, txHash };
}

export const MATERIAL_MAP = {
    energy:      { contributes: ['light', 'heat'], object: 'lamp' },
    computation: { contributes: ['signal', 'memory'], object: 'terminal' },
    structure:   { contributes: ['wall', 'floor'], object: 'shell' },
    reserve:     { contributes: ['continuity'], object: 'utility_box' }
} as const;

export function deriveRoom(events: VerifiedEvent[]): RoomState {
    // Rendering is reproducible. Model prose never changes room state.
    return events
        .filter(event => event.finalized && event.policyAccepted)
        .sort((a, b) => Number(a.blockNumber - b.blockNumber))
        .reduce(applyVerifiedMaterialChange, EMPTY_ROOM);
}
